import numpy as np
from scipy import ndimage
from geometry_msgs.msg import Pose, Point, Quaternion
from .base_explorer import BaseExplorer

class FrontierExplorer(BaseExplorer):
    """
    Frontier-based exploration algorithm.
    
    Identifies boundaries between known free space and unknown regions.
    Clusters frontiers and selects the nearest as next goal.
    
    Computational complexity: O(n) where n = grid cells
    CPU load: ~5-10%
    """
    
    def name(self) -> str:
        return "frontier"
    
    def compute_goal(self, density_map, robot_state, costmap):
        """Find nearest frontier and return as goal"""
        
        # Extract parameters
        min_frontier_size = self.params.get('min_frontier_size', 5)
        max_goal_distance = self.params.get('max_goal_distance', 20.0)
        
        # Find all frontier cells
        frontier_cells = self._find_frontier_cells(density_map)
        
        if not frontier_cells:
            self.logger.warn(f"{self.name()}: No frontiers found")
            return None
        
        # Cluster frontier cells
        frontier_clusters = self._cluster_frontiers(frontier_cells, density_map)
        
        # Filter by minimum size
        valid_clusters = [c for c in frontier_clusters 
                         if len(c) >= min_frontier_size]
        
        if not valid_clusters:
            self.logger.warn(f"{self.name()}: No frontiers above min size")
            return None
        
        # Find robot position
        robot_x, robot_y = robot_state['position']
        
        # Select nearest frontier cluster
        nearest_cluster = min(
            valid_clusters,
            key=lambda c: self._cluster_distance(c, robot_x, robot_y, density_map)
        )
        
        # Convert to world coordinates and create pose
        goal = self._cluster_to_pose(nearest_cluster, density_map)
        
        # Validate goal
        if not self.validate_goal(goal, costmap):
            self.logger.warn(f"{self.name()}: Goal validation failed")
            return None
        
        # Log success
        distance = self._cluster_distance(nearest_cluster, robot_x, robot_y, 
                                         density_map)
        self.logger.debug(
            f"{self.name()}: Selected goal at ({goal.position.x:.2f}, "
            f"{goal.position.y:.2f}), distance {distance:.2f}m"
        )
        
        return goal
    
    def _find_frontier_cells(self, density_map) -> list:
        """
        Find frontier cells: UNKNOWN cells adjacent to FREE cells
        
        Frontier = cell where occupancy is unknown (confidence low)
                   AND adjacent to a free cell (occupancy < 0.3)
        """
        frontier_cells = []
        
        for grid_y in range(1, density_map.grid_height - 1):
            for grid_x in range(1, density_map.grid_width - 1):
                occ = density_map.occupancy[grid_y, grid_x]
                conf = density_map.confidence[grid_y, grid_x]
                
                # Check if cell is unknown (low confidence)
                if conf < 0.3:
                    # Check if adjacent to free space
                    neighbors = [
                        (grid_x-1, grid_y-1), (grid_x, grid_y-1), (grid_x+1, grid_y-1),
                        (grid_x-1, grid_y),                         (grid_x+1, grid_y),
                        (grid_x-1, grid_y+1), (grid_x, grid_y+1), (grid_x+1, grid_y+1),
                    ]
                    
                    for nx, ny in neighbors:
                        if (0 <= nx < density_map.grid_width and 
                            0 <= ny < density_map.grid_height):
                            neighbor_occ = density_map.occupancy[ny, nx]
                            neighbor_conf = density_map.confidence[ny, nx]
                            
                            # Adjacent to free space with high confidence
                            if (neighbor_occ < 0.3 and neighbor_conf > 0.5):
                                frontier_cells.append((grid_x, grid_y))
                                break
        
        return frontier_cells
    
    def _cluster_frontiers(self, frontier_cells: list, 
                          density_map) -> list:
        """
        Cluster frontier cells using connected components.
        
        Returns:
            List of clusters, each cluster is a list of (grid_x, grid_y) tuples
        """
        if not frontier_cells:
            return []
        
        # Create binary map of frontier cells
        frontier_map = np.zeros((density_map.grid_height, 
                                density_map.grid_width), dtype=bool)
        for grid_x, grid_y in frontier_cells:
            frontier_map[grid_y, grid_x] = True
        
        # Label connected components
        labeled, num_clusters = ndimage.label(frontier_map)
        
        # Extract clusters
        clusters = []
        for cluster_id in range(1, num_clusters + 1):
            cluster_cells = np.where(labeled == cluster_id)
            clusters.append(list(zip(cluster_cells[1], cluster_cells[0])))
        
        return clusters
    
    def _cluster_distance(self, cluster: list, robot_x: float, 
                         robot_y: float, density_map) -> float:
        """Compute minimum distance from robot to cluster"""
        min_dist = float('inf')
        
        for grid_x, grid_y in cluster:
            world_x, world_y = density_map.grid_to_world(grid_x, grid_y)
            dist = ((world_x - robot_x)**2 + (world_y - robot_y)**2)**0.5
            min_dist = min(min_dist, dist)
        
        return min_dist
    
    def _cluster_to_pose(self, cluster: list, density_map) -> Pose:
        """Convert cluster to pose (at centroid)"""
        grid_x_coords = [x for x, y in cluster]
        grid_y_coords = [y for x, y in cluster]
        
        centroid_gx = np.mean(grid_x_coords)
        centroid_gy = np.mean(grid_y_coords)
        
        world_x, world_y = density_map.grid_to_world(int(centroid_gx), 
                                                     int(centroid_gy))
        
        return self.create_pose(world_x, world_y)

import numpy as np
import math
from geometry_msgs.msg import Pose
from .base_explorer import BaseExplorer
from .entropy_calculator import EntropyCalculator

class InfoGainExplorer(BaseExplorer):
    """
    Information gain-based exploration algorithm.
    
    Samples candidate locations and selects the one that maximizes
    expected information gain from future sensor observations.
    
    Computational complexity: O(k*r) where k = candidates, r = sensor range
    CPU load: ~15-25%
    """
    
    def name(self) -> str:
        return "info_gain"
    
    def compute_goal(self, density_map, robot_state, costmap):
        """Find goal that maximizes information gain"""
        
        # Extract parameters
        candidate_distances = self.params.get('candidate_distances', 
                                             [5.0, 10.0, 15.0])
        num_angles = self.params.get('num_angles', 16)
        sensor_range = self.params.get('lidar_max_range', 12.0)
        max_goal_distance = self.params.get('max_goal_distance', 20.0)
        
        robot_x, robot_y = robot_state['position']
        
        # Generate candidate locations
        candidates = self._generate_candidates(
            robot_x, robot_y, candidate_distances, num_angles, costmap
        )
        
        if not candidates:
            self.logger.warn(f"{self.name()}: No valid candidates generated")
            return None
        
        # Evaluate each candidate
        best_goal = None
        best_score = -float('inf')
        
        for candidate_x, candidate_y in candidates:
            # Compute information gain
            info_gain = EntropyCalculator.compute_info_gain_at_location(
                candidate_x, candidate_y, sensor_range, density_map
            )
            
            # Compute distance penalty (prefer closer goals)
            distance = ((candidate_x - robot_x)**2 + 
                       (candidate_y - robot_y)**2)**0.5
            distance_penalty = distance / max_goal_distance
            
            # Combine: maximize info gain, minimize distance
            score = info_gain - distance_penalty
            
            if score > best_score:
                best_score = score
                best_goal = (candidate_x, candidate_y)
        
        if best_goal is None:
            return None
        
        goal = self.create_pose(best_goal[0], best_goal[1])
        
        if not self.validate_goal(goal, costmap):
            self.logger.warn(f"{self.name()}: Goal validation failed")
            return None
        
        self.logger.debug(
            f"{self.name()}: Selected goal with info gain {best_score:.4f}"
        )
        
        return goal
    
    def _generate_candidates(self, robot_x: float, robot_y: float,
                            distances: list, num_angles: int,
                            costmap) -> list:
        """Generate circular ring of candidate locations"""
        candidates = []
        
        for distance in distances:
            for angle_idx in range(num_angles):
                angle = 2 * math.pi * angle_idx / num_angles
                
                # Candidate position
                candidate_x = robot_x + distance * math.cos(angle)
                candidate_y = robot_y + distance * math.sin(angle)
                
                # Validate (in free space)
                temp_pose = self.create_pose(candidate_x, candidate_y)
                if self.validate_goal(temp_pose, costmap):
                    candidates.append((candidate_x, candidate_y))
        
        return candidates

import numpy as np
import math
from geometry_msgs.msg import Pose
from .base_explorer import BaseExplorer
from .entropy_calculator import EntropyCalculator

class NBVExplorer(BaseExplorer):
    """
    Next-Best-View exploration algorithm.
    
    Sophisticated viewpoint selection considering occlusions and
    field-of-view constraints.
    
    Computational complexity: O(k*r^2) where k = candidates
    CPU load: ~30-50% (may approach budget limit)
    """
    
    def name(self) -> str:
        return "nbv"
    
    def compute_goal(self, density_map, robot_state, costmap):
        """Find optimal viewpoint for maximum information gain"""
        
        # Extract parameters
        candidate_distances = self.params.get('candidate_distances',
                                             [3.0, 6.0, 9.0, 12.0, 15.0])
        num_angles = self.params.get('num_angles', 32)
        sensor_range = self.params.get('lidar_max_range', 12.0)
        
        robot_x, robot_y = robot_state['position']
        
        # Generate candidates (more than info_gain)
        candidates = self._generate_candidates(
            robot_x, robot_y, candidate_distances, num_angles, costmap
        )
        
        if not candidates:
            self.logger.warn(f"{self.name()}: No valid candidates")
            return None
        
        # Evaluate with NBV computation
        best_goal = None
        best_score = -float('inf')
        
        for candidate_x, candidate_y in candidates:
            # Account for occlusions in visibility computation
            info_gain = self._compute_nbv_info_gain(
                candidate_x, candidate_y, sensor_range, density_map, costmap
            )
            
            distance = ((candidate_x - robot_x)**2 + 
                       (candidate_y - robot_y)**2)**0.5
            distance_penalty = distance / self.params.get('max_goal_distance', 20.0)
            
            score = info_gain - distance_penalty
            
            if score > best_score:
                best_score = score
                best_goal = (candidate_x, candidate_y)
        
        if best_goal is None:
            return None
        
        goal = self.create_pose(best_goal[0], best_goal[1])
        
        if not self.validate_goal(goal, costmap):
            return None
        
        self.logger.debug(
            f"{self.name()}: NBV goal with score {best_score:.4f}"
        )
        
        return goal
    
    def _compute_nbv_info_gain(self, viewpoint_x: float, viewpoint_y: float,
                              sensor_range: float, density_map, 
                              costmap) -> float:
        """
        Compute information gain accounting for occlusions.
        
        More sophisticated than simple range check: accounts for
        obstacles that block the view.
        """
        # Get cells in range
        cells = EntropyCalculator.get_cells_in_circle(
            viewpoint_x, viewpoint_y, sensor_range, density_map
        )
        
        total_gain = 0.0
        
        for grid_x, grid_y in cells:
            # Check if cell is visible (not occluded)
            if self._is_visible(viewpoint_x, viewpoint_y, grid_x, grid_y, 
                               density_map, costmap):
                
                prob = density_map.occupancy[grid_y, grid_x]
                confidence = density_map.confidence[grid_y, grid_x]
                
                if confidence < 0.8:
                    info_gain = EntropyCalculator.information_gain(prob, 1.0) \
                                if prob > 0.5 else EntropyCalculator.information_gain(prob, 0.0)
                    total_gain += info_gain
        
        return total_gain
    
    def _is_visible(self, viewer_x: float, viewer_y: float,
                   target_gx: int, target_gy: int,
                   density_map, costmap) -> bool:
        """
        Check if target cell is visible from viewpoint.
        
        Simple approximation: not visible if blocked by obstacle
        """
        target_x, target_y = density_map.grid_to_world(target_gx, target_gy)
        
        # Simple approach: check if there's a clear line of sight
        # (More sophisticated: full ray-tracing with 3D map)
        
        # For now: check if target occupancy is high (occluded)
        if density_map.occupancy[target_gy, target_gx] > 0.7:
            return False
        
        return True
    
    def _generate_candidates(self, robot_x: float, robot_y: float,
                            distances: list, num_angles: int,
                            costmap) -> list:
        """Generate candidates with finer resolution"""
        candidates = []
        
        for distance in distances:
            for angle_idx in range(num_angles):
                angle = 2 * math.pi * angle_idx / num_angles
                
                candidate_x = robot_x + distance * math.cos(angle)
                candidate_y = robot_y + distance * math.sin(angle)
                
                temp_pose = self.create_pose(candidate_x, candidate_y)
                if self.validate_goal(temp_pose, costmap):
                    candidates.append((candidate_x, candidate_y))
        
        return candidates

import rclpy
from rclpy.node import Node
from rclpy.timer import Timer
from nav_msgs.msg import Odometry, OccupancyGrid
from geometry_msgs.msg import PoseStamped
from sensor_msgs.msg import LaserScan
import tf2_ros
import tf2_geometry_msgs
import yaml

from .density_map import DensityMap
from .entropy_calculator import EntropyCalculator
from .frontier_explorer import FrontierExplorer
from .info_gain_explorer import InfoGainExplorer
from .nbv_explorer import NBVExplorer

class ExplorationNode(Node):
    """
    Main ROS2 node for D2OC exploration.
    
    Manages sensor subscriptions, density map updates, and algorithm
    execution to generate exploration goals.
    """
    
    def __init__(self):
        super().__init__('exploration_node')
        
        # Declare parameters
        self.declare_parameter('algorithm', 'frontier')
        self.declare_parameter('config_file', '')
        self.declare_parameter('grid_width', 100.0)
        self.declare_parameter('grid_height', 100.0)
        self.declare_parameter('grid_resolution', 0.2)
        self.declare_parameter('control_frequency', 1.0)
        self.declare_parameter('enable_visualization', True)
        self.declare_parameter('enable_scan_update', True)
        
        # Get parameters
        algorithm_name = self.get_parameter('algorithm').value
        config_file = self.get_parameter('config_file').value
        self.grid_width = self.get_parameter('grid_width').value
        self.grid_height = self.get_parameter('grid_height').value
        self.grid_resolution = self.get_parameter('grid_resolution').value
        self.control_frequency = self.get_parameter('control_frequency').value
        self.enable_visualization = self.get_parameter('enable_visualization').value
        self.enable_scan_update = self.get_parameter('enable_scan_update').value
        
        self.get_logger().info(f"Initializing exploration node with {algorithm_name}")
        
        # Load configuration
        params = self._load_config(config_file) if config_file else {}
        
        # Initialize density map
        self.density_map = DensityMap(
            width=self.grid_width,
            height=self.grid_height,
            resolution=self.grid_resolution
        )
        self.get_logger().info(
            f"Density map: {self.grid_width}x{self.grid_height}m "
            f"@ {self.grid_resolution}m resolution"
        )
        
        # Initialize algorithm
        algorithm_map = {
            'frontier': FrontierExplorer,
            'info_gain': InfoGainExplorer,
            'nbv': NBVExplorer,
        }
        
        if algorithm_name not in algorithm_map:
            raise ValueError(f"Unknown algorithm: {algorithm_name}")
        
        self.explorer = algorithm_map[algorithm_name](self, params)
        self.get_logger().info(f"Using {self.explorer.name()} explorer")
        
        # Initialize TF2 buffer for transforms
        self.tf_buffer = tf2_ros.Buffer()
        self.tf_listener = tf2_ros.TransformListener(self.tf_buffer, self)
        
        # Subscribers
        self.scan_sub = self.create_subscription(
            LaserScan, '/scan', self.scan_callback, 10
        )
        
        self.odom_sub = self.create_subscription(
            Odometry, '/odometry/filtered', self.odom_callback, 10
        )
        
        self.costmap_sub = self.create_subscription(
            OccupancyGrid, '/local_costmap/costmap', self.costmap_callback, 10
        )
        
        # Publishers
        self.goal_pub = self.create_publisher(
            PoseStamped, '/exploration/goal', 10
        )
        
        if self.enable_visualization:
            self.density_map_pub = self.create_publisher(
                OccupancyGrid, '/d2oc/density_map', 10
            )
        
        # Control timer
        period = 1.0 / self.control_frequency
        self.timer = self.create_timer(period, self.exploration_loop)
        
        # State
        self.robot_state = None
        self.costmap = None
        self.latest_scan = None
        self.goal_count = 0
        
        self.get_logger().info("Exploration node ready")
    
    def _load_config(self, config_file: str) -> dict:
        """Load YAML configuration file"""
        try:
            with open(config_file, 'r') as f:
                return yaml.safe_load(f) or {}
        except Exception as e:
            self.get_logger().warn(f"Failed to load config {config_file}: {e}")
            return {}
    
    def scan_callback(self, msg: LaserScan):
        """Process LiDAR scan and update density map"""
        self.latest_scan = msg
        
        if not self.enable_scan_update or self.robot_state is None:
            return
        
        try:
            # Get robot pose from TF
            transform = self.tf_buffer.lookup_transform(
                'map', 'base_link', rclpy.time.Time()
            )
            
            robot_x = transform.transform.translation.x
            robot_y = transform.transform.translation.y
            
            # Extract orientation (quaternion to angle)
            q = transform.transform.rotation
            import math
            robot_theta = math.atan2(
                2*(q.w*q.z + q.x*q.y),
                1 - 2*(q.y*q.y + q.z*q.z)
            )
            
            # Update density map
            self.density_map.update_from_scan(
                msg, robot_x, robot_y, robot_theta,
                confidence=0.9,
                timestamp=self.get_clock().now()
            )
            
        except Exception as e:
            self.get_logger().debug(f"TF lookup failed: {e}")
    
    def odom_callback(self, msg: Odometry):
        """Store robot state"""
        self.robot_state = {
            'position': (
                msg.pose.pose.position.x,
                msg.pose.pose.position.y
            ),
            'orientation': msg.pose.pose.orientation,
            'velocity': (
                msg.twist.twist.linear.x,
                msg.twist.twist.angular.z
            ),
        }
    
    def costmap_callback(self, msg: OccupancyGrid):
        """Store costmap and update density if needed"""
        self.costmap = msg
        
        # Optionally integrate costmap into density map
        # (Comment out if causing issues)
        # self.density_map.update_from_costmap(msg)
    
    def exploration_loop(self):
        """Main exploration control loop"""
        
        # Check prerequisites
        if self.robot_state is None:
            self.get_logger().debug("Waiting for robot state")
            return
        
        if self.costmap is None:
            self.get_logger().debug("Waiting for costmap")
            return
        
        # Compute next goal
        goal_pose = self.explorer.compute_goal(
            self.density_map,
            self.robot_state,
            self.costmap
        )
        
        # Publish goal if valid
        if goal_pose is not None:
            self._publish_goal(goal_pose)
            self.goal_count += 1
        else:
            self.get_logger().debug(f"{self.explorer.name()}: No valid goal")
        
        # Optionally publish density map for visualization
        if self.enable_visualization:
            self._publish_density_map()
    
    def _publish_goal(self, goal_pose):
        """Publish exploration goal"""
        msg = PoseStamped()
        msg.header.frame_id = 'map'
        msg.header.stamp = self.get_clock().now().to_msg()
        msg.pose = goal_pose
        
        self.goal_pub.publish(msg)
        
        self.get_logger().info(
            f"[{self.goal_count}] Goal: ({goal_pose.position.x:.2f}, "
            f"{goal_pose.position.y:.2f}) [{self.explorer.name()}]"
        )
    
    def _publish_density_map(self):
        """Publish density map for RViz visualization"""
        try:
            msg = self.density_map.to_occupancy_grid(
                frame_id='map',
                timestamp=self.get_clock().now().to_msg()
            )
            self.density_map_pub.publish(msg)
        except Exception as e:
            self.get_logger().debug(f"Failed to publish density map: {e}")


def main():
    rclpy.init()
    node = ExplorationNode()
    rclpy.spin(node)
    node.destroy_node()
    rclpy.shutdown()

if __name__ == '__main__':
    main()

APPROACH: DUAL-GOAL SYSTEM

Current Architecture:
Nav2 receives goals from two sources:
1. Traditional waypoint navigation (existing)
2. D2OC exploration goals (new)

Integration Method:
- D2OC publishes to /exploration/goal (new topic)
- Nav2 continues publishing to /goal_pose
- High-level navigation can choose which to pursue
- Recommended: Use twist_mux or create wrapper

Option 1: SIMPLE INTEGRATION (Recommended for testing)
======================================================
Run D2OC alongside Nav2, manually feed goals:

Launch sequence:
1. Start Nav2 normally:
   $ ros2 launch robot_navigation navigation_launch.py

2. In separate terminal, start D2OC:
